+import { Pool } from "pg";
+import { PgBoss, Job } from "pg-boss";
+import { env } from "../env.js";
+import { fileURLToPath } from "node:url";
+import { WorkflowRunner } from "../../../../packages/orchestrator/src/workflow/runner.js";
+import { Registry } from "../../../../packages/orchestrator/src/workflow/registry.js";
+import { DBRunbookResolver } from "../../../../packages/orchestrator/src/resolver.js";
+import type { EventEnvelope } from "@tekisho/domain";
+
+export const DISPATCH_QUEUE = "orchestration.dispatch";
+
+export async function startDispatcherWorker(): Promise<() => Promise<void>> {
+  const pool = new Pool({
+    connectionString: env.DATABASE_MIGRATOR_URL,
+    ssl: { rejectUnauthorized: false },
+  });
+
+  const boss = new PgBoss({
+    connectionString: env.DATABASE_MIGRATOR_URL,
+    schema: "pgboss",
+    ssl: { rejectUnauthorized: false },
+  });
+
+  await boss.start();
+
+  const resolver = new DBRunbookResolver(pool);
+  const registry = new Registry();
+  const runner = new WorkflowRunner(resolver, registry, pool);
+
+  await boss.work<EventEnvelope>(DISPATCH_QUEUE, async ([job]) => {
+    try {
+      const envelope = job.data;
+      if (!envelope || !envelope.event_type || !envelope.event_id) {
+        throw new Error(`Invalid event envelope in job ${job.id}`);
+      }
+
+      console.log(`[dispatcher] Processing event ${envelope.event_id} (${envelope.event_type})`);
+
+      const result = await runner.executeWorkflow({
+        event_id: envelope.event_id,
+        event_type: envelope.event_type,
+        aggregate_id: envelope.aggregate.id,
+        payload: envelope.payload_ref || {}
+      });
+
+      if (result.status === 'FAILED') {
+        throw new Error(`Workflow execution failed: ${result.error}`);
+      }
+      
+      console.log(`[dispatcher] Event ${envelope.event_id} COMPLETED`);
+    } catch (error: any) {
+      console.error(`[dispatcher] Error processing job ${job.id}:`, error);
+      throw error;
+    }
+  });
+
+  console.log(`Dispatcher worker started → queue "${DISPATCH_QUEUE}"`);
+
+  return async () => {
+    await boss.stop();
+    await pool.end();
+    console.log("Dispatcher worker stopped");
+  };
+}
+
+const isMainModule = process.argv[1] && fileURLToPath(import.meta.url) === process.argv[1];
+if (isMainModule) {
+  startDispatcherWorker()
+    .then((stop) => {
+      const shutdown = () => void stop().then(() => process.exit(0));
+      process.on("SIGINT", shutdown);
+      process.on("SIGTERM", shutdown);
+    })
+    .catch((error) => {
+      console.error("Failed to start dispatcher:", error);
+      process.exitCode = 1;
+    });
+}

+# Orchestration Flow Investigation Report
+
+## 1. Source for Picture C (jd.uploaded, etc.)
+**Zero call sites found.** 
+I searched the entire monorepo for `jd.uploaded`, `resume.uploaded`, `match.requested`, `runForJob`, `runForResume`, `topics.js`, and `eventTypes.js`. 
+- There are no `boss.send()` or `boss.work()` calls for those queues.
+- `apps/api/src/index.ts` is an empty stub (`// TODO: Add the Fastify API entry point.`). There are no `POST /jobs` or `POST /resumes` route handlers.
+- **Conclusion**: Picture C physically does not exist in this monorepo. It describes a completely different or legacy codebase.
+
+## 2. event_outbox Writers
+**Zero call sites found.**
+I grepped the entire monorepo for `INSERT INTO event_outbox`. 
+- The table is created in migrations (`packages/db/migrations/V3 — Agentic spine.sql`).
+- It is selected/updated by the relay (`packages/orchestrator/src/relay/relay.ts`).
+- There is absolutely no application code writing to this table in the live flow today.
+
+## 3. orchestration.dispatch Consumer
+**A consumer exists.**
+`apps/worker/src/bootstrap-pgboss.ts` consumes this queue.
+```typescript
+// apps/worker/src/bootstrap-pgboss.ts:13-17
+await boss.work('orchestration.dispatch', {
+  teamSize: 5, // Concurrency support
+  teamConcurrency: 5,
+  newJobCheckInterval: 500
+}, async (job) => {
+```
+
+## 4. packages/orchestrator Imports
+**It is imported by the worker.**
+Outside of tests, it is imported by `apps/worker/src/bootstrap-pgboss.ts`:
+```typescript
+// apps/worker/src/bootstrap-pgboss.ts:3-4
+import type { PolicyEngine } from '@tekisho/orchestrator/src/policy/engine';
+import { runWorkflow } from '@tekisho/orchestrator/src/workflow/runner';
+```
+*(Note: These imports use deep paths into `/src/` instead of the package root exports, but they are physically there).*
+
+## 5. Are these pictures connected?
+**Picture A and Picture B are wired together in a vacuum, but are 100% DISCONNECTED from Picture C.**
+Picture C does not exist here. Picture A (the orchestrator) and Picture B (the outbox and worker) are connected to each other: the relay reads from `event_outbox` -> pushes to pg-boss `orchestration.dispatch` -> the worker consumes it -> calls the orchestrator `runWorkflow`. However, because **nothing** writes to `event_outbox`, this entire engine is currently an idling, isolated island of code with zero live traffic.
+

+
+### Diagram: What ACTUALLY happens today when a JD or resume is uploaded
+```
+[JD / Resume Upload]
+       |
+       v
+  (Nothing happens. The API routes do not exist in this repo.)
+```
+
+### If I publish an event shaped like 'requirement.created' or 'syntra.jd.received' right now, does anything react to it?
+**No.** 
+If you simply publish it as an in-memory event, nothing reacts. However, if you manually insert a row into the `event_outbox` database table (and the corresponding `events` table), the `OutboxRelay` *would* pick it up, send it to pg-boss, and the worker *would* pass it to the `PolicyEngine`.
+
+### Shortest path to participate in a real JD upload today
+Because the "real" JD upload API routes (Picture C) do not exist in this monorepo, the shortest path requires two things:
+1. **Rebuild the missing API**: You must write the `POST /jobs` and `POST /resumes` route handlers in `apps/api`.
+2. **Write to the outbox**: Those route handlers must perform a transactional `INSERT INTO events` and `INSERT INTO event_outbox`. Once they do that, the existing relay, worker, and orchestrator (Pictures A & B) will automatically wake up and process the events.

+import type { Pool } from 'pg';
+
+export interface GraphDefinition {
+  id: string;
+  org_id: string | null;
+  graph_key: string;
+  version: number;
+  definition: Record<string, any>;
+  checksum: string;
+  status: string;
+}
+
+export class DBRunbookResolver {
+  private pool: Pool;
+  
+  private eventToGraphMap: Record<string, string> = {
+    'jd.uploaded': 'syntra.jd.workflow',
+    'resume.uploaded': 'syntra.resume.workflow',
+    'match.requested': 'syntra.match.workflow'
+  };
+
+  constructor(pool: Pool) {
+    this.pool = pool;
+  }
+
+  public async resolve(eventType: string, orgId?: string): Promise<GraphDefinition> {
+    const graphKey = this.eventToGraphMap[eventType];
+    if (!graphKey) {
+      throw new Error(`No workflow mapping found for event type: ${eventType}`);
+    }
+
+    const { rows } = await this.pool.query(
+      `SELECT id, org_id, graph_key, version, definition, checksum, status
+       FROM graph_definitions
+       WHERE graph_key = $1 AND status = 'active'
+       ORDER BY org_id NULLS LAST`,
+      [graphKey]
+    );
+
+    if (rows.length === 0) {
+      throw new Error(`No active graph definition found for ${graphKey}`);
+    }
+
+    // Filter by orgId if provided, else fallback to global (org_id IS NULL)
+    let activeDefs = rows;
+    if (orgId) {
+      const orgSpecific = rows.filter(r => r.org_id === orgId);
+      if (orgSpecific.length > 0) {
+        activeDefs = orgSpecific;
+      } else {
+        activeDefs = rows.filter(r => r.org_id === null);
+      }
+    } else {
+      activeDefs = rows.filter(r => r.org_id === null);
+    }
+
+    if (activeDefs.length === 0) {
+      throw new Error(`No active graph definition found for ${graphKey} and org ${orgId || 'global'}`);
+    }
+
+    if (activeDefs.length > 1) {
+      throw new Error(`Multiple active graph definitions found for ${graphKey}. Ambiguous resolution.`);
+    }
+
+    return activeDefs[0];
+  }
+}

+import { StateGraph, Annotation, type CompiledGraph } from '@langchain/langgraph';
+import { Registry } from './registry.js';
+import type { GraphDefinition } from '../resolver.js';
+import type { EventInput } from '../types.js';
+
+export const SyntraState = Annotation.Root({
+  trigger_event: Annotation<EventInput>({
+    reducer: (x, y) => y ?? x,
+    default: () => ({ event_id: '', event_type: '', aggregate_id: '', payload: {} })
+  }),
+  syntraResult: Annotation<any>({
+    reducer: (x, y) => y ?? x,
+    default: () => null
+  }),
+  _emittedEvents: Annotation<EventInput[]>({
+    reducer: (x, y) => x.concat(y || []),
+    default: () => []
+  })
+});
+
+export class GraphCompiler {
+  public static compile(graphDef: GraphDefinition): CompiledGraph<any, any, any> {
+    const { graph_key, definition } = graphDef;
+
+    if (!definition || !Array.isArray(definition.nodes) || !Array.isArray(definition.edges)) {
+      throw new Error(`Invalid graph definition for ${graph_key}: Missing nodes or edges array`);
+    }
+
+    const graph = new StateGraph(SyntraState);
+
+    for (const node of definition.nodes) {
+      if (!node.id || !node.impl) {
+        throw new Error(`Invalid node in ${graph_key}: Missing id or impl`);
+      }
+      
+      const implFunction = Registry.getNodeImplementation(node.impl);
+      graph.addNode(node.id, implFunction);
+    }
+
+    for (const edge of definition.edges) {
+      if (!Array.isArray(edge) || edge.length !== 2) {
+        throw new Error(`Invalid edge in ${graph_key}: Edge must be a tuple [from, to]`);
+      }
+      graph.addEdge(edge[0], edge[1]);
+    }
+
+    // Default checkpointing is NOT wired here yet per Phase 23 constraint.
+    return graph.compile();
+  }
+}

+function getApiUrl(): string {

+async function safeFetch(url: string, options: RequestInit): Promise<any> {
+  const controller = new AbortController();
+  const timeoutId = setTimeout(() => controller.abort(), 35000); // 35s timeout
+
+  try {
+    const response = await fetch(url, {
+      ...options,
+      signal: controller.signal as any,
+    });
+
+    const text = await response.text();
+    let data;
+    try {
+      data = text ? JSON.parse(text) : {};
+    } catch {
+      data = { raw: text };
+    }
+
+    if (!response.ok) {
+      const detail = data?.error || data?.detail || text || response.statusText;
+      const err = new Error(`HTTP ${response.status}: ${detail}`);
+      (err as any).statusCode = response.status;
+      throw err;
+    }
+
+    return data;
+  } catch (error: any) {
+    if (error.name === 'AbortError') {
+      throw new Error(`Syntra API Timeout (${url})`);
+    }
+    throw error;
+  } finally {
+    clearTimeout(timeoutId);
+  }
+}
